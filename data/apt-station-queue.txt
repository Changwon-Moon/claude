# kapt 는 세대수·법정동 대조로 짚었다(아래 주석에 근거).

kapt=A44340013 force=1

kapt=A13686302
# 화서역푸르지오더에듀포레 — 로그 세대수 2571 · 천천동 333
kapt=A44033010
# 서초포레스타2단지 — 로그 세대수 1077 · 내곡동 384
kapt=A10028021 force=1
# 다산롯데캐슬 — 로그 세대수 1186 · 다산동 6029
kapt=A10026600
